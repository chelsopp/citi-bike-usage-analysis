# Create the cleaned dataset expected later in the analysis

citibike_2023_clean <- citibike_2023 %>%
  mutate(
    # ride length
    ride_length_min = as.numeric(difftime(ended_at, started_at, units = "mins")),
    
    # Date/time components
    start_date = as_date(started_at),
    start_hour = hour(started_at),
    start_month = month(started_at, label = TRUE, abbr = FALSE),
    start_day = wday(started_at, label = TRUE, abbr = FALSE),
    start_year = year(started_at),

    # time of day category
    time_of_day = case_when(
      start_hour >= 5 & start_hour < 12 ~ "Morning",
      start_hour >= 12 & start_hour < 17 ~ "Afternoon",
      start_hour >= 17 & start_hour < 21 ~ "Evening",
      TRUE ~ "Night"
    ),

    # bike type cleaned
    bike_type = case_when(
      rideable_type == "electric_bike" ~ "Electric",
      rideable_type == "classic_bike"  ~ "Classic",
      rideable_type == "docked_bike"   ~ "Docked",
      TRUE ~ "Other"
    ),

    # weekend vs weekday
    day_type = ifelse(start_day %in% c("Saturday", "Sunday"), "Weekend", "Weekday")
  ) %>%
  # remove problematic/unrealistic rides
  filter(
    ride_length_min >= 1,
    ride_length_min <= 24 * 60,
    started_at < ended_at,
    !(start_station_id == end_station_id & ride_length_min < 2)
  )

cat("Created citibike_2023_clean with", format(nrow(citibike_2023_clean), big.mark=","), "rows\n")

# Use our existing cleaned NJ dataset and adding NJ-specific variables
citibike_nj_analysis <- citibike_2023_clean %>%
  # Add NJ-specific variables
  mutate(
    # Create consistent member_type variable
    member_type = ifelse(member_casual == "member", "Annual Member", "Casual Rider"),
    member_type = factor(member_type, levels = c("Annual Member", "Casual Rider")),
    
    # Create clean time_of_day with proper ordering
    time_of_day = factor(time_of_day, 
                         levels = c("Morning", "Afternoon", "Evening", "Night")),
    
    # Create clean bike_type with proper ordering
    bike_type = factor(bike_type, 
                       levels = c("Classic", "Electric", "Docked", "Other")),
    
    # Add season variable for later use
    season = case_when(
      start_month %in% c("December", "January", "February") ~ "Winter",
      start_month %in% c("March", "April", "May") ~ "Spring",
      start_month %in% c("June", "July", "August") ~ "Summer",
      start_month %in% c("September", "October", "November") ~ "Fall"
    ),
    season = factor(season, levels = c("Winter", "Spring", "Summer", "Fall")),
    
    # Add day of week for heatmap
    day_of_week = wday(started_at, label = TRUE, abbr = TRUE),
    
    # Add ride length categories for easier analysis
    ride_length_category = case_when(
      ride_length_min < 10 ~ "Short (<10 min)",
      ride_length_min < 30 ~ "Medium (10-30 min)",
      ride_length_min < 60 ~ "Long (30-60 min)",
      TRUE ~ "Very Long (>60 min)"
    ),
    ride_length_category = factor(ride_length_category,
                                  levels = c("Short (<10 min)", "Medium (10-30 min)", 
                                             "Long (30-60 min)", "Very Long (>60 min)")),
    
    # Add NJ area categories based on longitude
    nj_area = case_when(
      start_lng < -74.03 ~ "Jersey City",
      start_lng >= -74.03 ~ "Hoboken",
      TRUE ~ "Other NJ Area"
    ),
    nj_area = factor(nj_area, levels = c("Hoboken", "Jersey City", "Other NJ Area"))
  )
